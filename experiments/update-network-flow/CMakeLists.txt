cmake_minimum_required(VERSION 3.24)
project(xdnote_update_network_flow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Core Network)
add_executable(update_network_flow flow.cpp)
target_link_libraries(update_network_flow PRIVATE Qt6::Core Qt6::Network ${CMAKE_DL_LIBS})
add_executable(update_network_socket_bounds socket_bounds.cpp)
target_link_libraries(update_network_socket_bounds PRIVATE Qt6::Core Qt6::Network Qt6::NetworkPrivate)
if(APPLE)
  target_compile_options(update_network_flow PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/../../cmake/QtCompilerCompatibility.h")
  set_target_properties(update_network_flow PROPERTIES OSX_ARCHITECTURES arm64 OSX_DEPLOYMENT_TARGET 14.0)
  target_compile_options(update_network_socket_bounds PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/../../cmake/QtCompilerCompatibility.h")
  set_target_properties(update_network_socket_bounds PROPERTIES OSX_ARCHITECTURES arm64 OSX_DEPLOYMENT_TARGET 14.0)
endif()
