cmake_minimum_required(VERSION 3.24)
project(xdnote_native_preview_evaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Widgets)
if(NOT EXISTS "${XDNOTE_RUST_LIBRARY}")
    message(FATAL_ERROR "Provide this project's identity-verified Rust static library")
endif()
add_executable(preview_native probe.cpp ../../apps/desktop/src/UiFonts.cpp)
target_include_directories(preview_native PRIVATE ../../apps/desktop/src)
target_link_libraries(preview_native PRIVATE Qt6::Widgets "${XDNOTE_RUST_LIBRARY}")
if(WIN32)
    target_link_libraries(preview_native PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
    target_compile_options(preview_native PRIVATE /utf-8 /EHsc)
endif()
