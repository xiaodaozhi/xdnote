cmake_minimum_required(VERSION 3.24)
project(xdnote_preview_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Widgets WebEngineWidgets)
if(NOT EXISTS "${XDNOTE_RUST_LIBRARY}")
    message(FATAL_ERROR "Provide the current project's already built Rust static library")
endif()
foreach(kind native webengine)
    add_executable(preview_${kind} probe.cpp)
    target_include_directories(preview_${kind} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/desktop/src")
    target_link_libraries(preview_${kind} PRIVATE Qt6::Widgets "${XDNOTE_RUST_LIBRARY}")
    if(WIN32)
        target_link_libraries(preview_${kind} PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
        target_compile_options(preview_${kind} PRIVATE /utf-8 /EHsc)
    endif()
endforeach()
target_compile_definitions(preview_webengine PRIVATE WITH_WEBENGINE)
target_link_libraries(preview_webengine PRIVATE Qt6::WebEngineWidgets)
