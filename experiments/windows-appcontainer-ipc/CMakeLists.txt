cmake_minimum_required(VERSION 3.20)
project(xdnote_ipc_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Core Network)
add_executable(ipc_probe main.cpp ../../apps/desktop/src/SingleInstance.cpp)
target_include_directories(ipc_probe PRIVATE ../../apps/desktop/src)
target_link_libraries(ipc_probe PRIVATE Qt6::Core Qt6::Network)
if(WIN32)
  target_link_libraries(ipc_probe PRIVATE advapi32)
endif()
add_executable(container_isolation_probe container_isolation.cpp)
target_link_libraries(container_isolation_probe PRIVATE advapi32)
add_executable(product_instance_probe product_instance_probe.cpp ../../apps/desktop/src/SingleInstance.cpp)
target_include_directories(product_instance_probe PRIVATE ../../apps/desktop/src)
target_link_libraries(product_instance_probe PRIVATE Qt6::Core Qt6::Network advapi32)
target_compile_definitions(product_instance_probe PRIVATE XDNOTE_IPC_TEST_HOOKS)
if(MSVC)
  target_compile_options(ipc_probe PRIVATE /utf-8)
  target_compile_options(container_isolation_probe PRIVATE /utf-8)
  target_compile_options(product_instance_probe PRIVATE /utf-8)
endif()
