cmake_minimum_required(VERSION 3.20)
project(xdnote_ipc_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.2 EXACT REQUIRED COMPONENTS Core Network)
add_executable(ipc_probe main.cpp ../../apps/desktop/src/SingleInstance.cpp)
target_include_directories(ipc_probe PRIVATE ../../apps/desktop/src)
target_link_libraries(ipc_probe PRIVATE Qt6::Core Qt6::Network)
if(MSVC)
  target_compile_options(ipc_probe PRIVATE /utf-8)
endif()
