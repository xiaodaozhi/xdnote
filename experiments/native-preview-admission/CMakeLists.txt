cmake_minimum_required(VERSION 3.24)
project(xdnote_native_preview_admission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../tools/desktop-dependencies.json" XDNOTE_DEPENDENCIES)
string(JSON XDNOTE_QT_VERSION GET "${XDNOTE_DEPENDENCIES}" qt version)
find_package(Qt6 ${XDNOTE_QT_VERSION} EXACT REQUIRED COMPONENTS Widgets Svg Concurrent Test)
if(NOT EXISTS "${XDNOTE_RUST_LIBRARY}")
    message(FATAL_ERROR "Provide this checkout's freshly built Rust library")
endif()
add_library(xdnote_markdown STATIC IMPORTED GLOBAL)
set_target_properties(xdnote_markdown PROPERTIES IMPORTED_LOCATION "${XDNOTE_RUST_LIBRARY}")
include(../../cmake/NativePreviewModel.cmake)
include(../../cmake/NativePreviewView.cmake)
add_executable(xdnote_native_preview_probe probe.cpp)
target_link_libraries(xdnote_native_preview_probe PRIVATE xdnote_native_preview xdnote_native_preview_test_ui)
if(MSVC)
    target_compile_options(xdnote_native_preview_probe PRIVATE /utf-8 /EHsc)
endif()
